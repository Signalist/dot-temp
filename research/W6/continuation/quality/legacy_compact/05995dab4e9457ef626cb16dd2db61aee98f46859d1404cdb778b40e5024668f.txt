```python
def count_elements(lst):
    """
    Count the number of elements in the list.

    Parameters:
    lst (list): The list of integers to count elements in.

    Returns:
    int: The count of elements in the list.
    """
    return len(lst)

def filter_elements(lst, condition):
    """
    Filter elements in the list based on a condition.

    Parameters:
    lst (list): The list of integers to filter.
    condition (function): A function that takes an element of the list and returns True if the element should be included in the result, False otherwise.

    Returns:
    list: A list containing only the elements that satisfy the condition.
    """
    return [element for element in lst if condition(element)]

def sort_elements(lst):
    """
    Sort the list in ascending order.

    Parameters:
    lst (list): The list of integers to sort.

    Returns:
    list: The sorted list.
    """
    return sorted(lst)

def sum_elements(lst):
    """
    Sum the elements in the list.

    Parameters:
    lst (list): The list of integers to sum.

    Returns:
    int: The sum of the elements in the list.
    """
    return sum(lst)

def find_runs(lst):
    """
    Find all runs in the list.

    Parameters:
    lst (list): The list of integers to find runs in.

    Returns:
    list: A list of runs, where each run is a list of consecutive integers.
    """
    runs = []
    current_run = []
    for i in range(len(lst) - 1):
        if lst[i] + 1 == lst[i + 1]:
            current_run.append(lst[i])
        else:
            if current_run:
                runs.append(current_run)
            current_run = [lst[i]]
    if current_run:
        runs.append(current_run)
    return runs
```
